# 扩展中国剩余定理（EXCRT）

## 模板用途

合并模数不一定互质的多组同余方程，并求最小非负解。

## 核心思路

- 设当前解为 `x ≡ a1 (mod n1)`，新方程为 `x ≡ a2 (mod n2)`。
- 转化为线性方程 `n1·t ≡ a2-a1 (mod n2)`。
- 使用扩展欧几里得判断是否有解；有解时将两个方程合并成一个模最小公倍数的新方程。

## 复杂度分析

合并一个方程的复杂度为 `O(log M)`；合并 `k` 个方程为 `O(klog M)`，空间复杂度为 `O(1)`。

## 使用提示

这是学习模板，使用时需要根据具体题目补充节点数、边数、数组范围、输入输出以及必要的初始化。模板文件中若包含多个相近算法，应按对应小节选择其中一种实现，而不是把多个独立程序同时编译。

#include <bits/stdc++.h>

long long exgcd(long long a,long long b,long long &x,long long &y){
    if(!b){x=1;y=0;return a;}
    long long d=exgcd(b,a%b,y,x);
    y-=a/b*x;
    return d;
}

// �ϲ� m �� a1 (mod n1), m �� a2 (mod n2)
bool merge(long long a1, long long n1, long long a2, long long n2, long long &a3, long long &n3){
    long long x, y;
    long long d = exgcd(n1, n2, x, y);
    long long c = a2 - a1;
    if (c % d != 0) return false; // �޽�
    n3 = n1 / d * n2;
    x = x * (c / d) % (n2 / d);
    a3 = (a1 + n1 * x) % n3;
    if (a3 < 0) a3 += n3;
    return true;
}

int main(){
    int n;
    string s;
    cin >> n >> s;
    vector<char> seq(n);
    for(int i=0;i<n;i++) seq[i]='A'+i;

    long long a = 0, mod = 1;
    for(int i=0;i<n;i++){
        auto it=find(seq.begin(), seq.end(), s[i]);
        int idx =distance(seq.begin(), it);
        // ��ǰ��Ҫ�� m % (n-i) == idx
        long long new_a, new_mod;
        if (!merge(a, mod, idx, n-i, new_a, new_mod)){
            cout << "NO\n";
            return 0;
        }
        a = new_a;
        mod = new_mod;
        seq.erase(seq.begin()+idx);
    }
    cout << "YES\n" << a << "\n";
    return 0;
}
