# 线性同余方程与中国剩余定理

## 模板用途

求解线性同余方程，并逐步合并多组同余限制。

## 核心思路

- 线性同余方程 `ax ≡ b (mod n)` 有解当且仅当 `gcd(a,n)` 整除 `b`。
- 使用扩展欧几里得算法求系数及模意义下的解。
- 多组同余条件可以逐个合并；模数不互质时需要先检查兼容性。

## 复杂度分析

每次扩展欧几里得的复杂度为 `O(log n)`；逐个合并 `k` 个方程约为 `O(klog M)`。

## 使用提示

这是学习模板，使用时需要根据具体题目补充节点数、边数、数组范围、输入输出以及必要的初始化。模板文件中若包含多个相近算法，应按对应小节选择其中一种实现，而不是把多个独立程序同时编译。

## code

```C++
#include <iostream>
#include <vector>
#include <string>
using namespace std;

// Extended Euclidean Algorithm to solve ax �� b mod n for x
long long solve_mod(long long a, long long b, long long n) {
    a = (a % n + n) % n;
    b = (b % n + n) % n;
    if (a == 0) {
        return b == 0 ? 0 : -1; // No solution unless b == 0
    }
    // ax �� b (mod n)
    long long x, y, g;
    // Extended GCD
    auto ext_gcd = [](long long a, long long b, long long &x, long long &y) {
        if (b == 0) { x = 1; y = 0; return a; }
        long long x1, y1;
        long long d = ext_gcd(b, a % b, x1, y1);
        x = y1;
        y = x1 - (a / b) * y1;
        return d;
    };
    g = ext_gcd(a, n, x, y);
    if (b % g != 0) return -1; // No solution
    x = x * (b / g) % n;
    x = (x + n) % n;
    return x;
}

int main() {
    int n;
    string target;
    cin >> n >> target;
    vector<char> alpha(n);
    for (int i = 0; i < n; ++i) alpha[i] = 'A' + i;
    vector<int> mod_seq; // stores the mod indices at each step

    vector<char> cur = alpha;
    for (int i = 0; i < n; ++i) {
        auto it = find(cur.begin(), cur.end(), target[i]);
        if (it == cur.end()) {
            cout << "NO\n";
            return 0;
        }
        int idx = it - cur.begin();
        mod_seq.push_back(idx);
        cur.erase(cur.begin() + idx);
    }

    // Now, solve for m such that:
    // m mod n == mod_seq[0]
    // m mod (n-1) == mod_seq[1]
    // ...
    // m mod 1 == mod_seq[n-1]
    // Use Chinese Remainder Theorem

    long long M = 0;
    long long step = 1;
    bool ok = true;
    for (int i = n - 1; i >= 0; --i) {
        int mod_base = i + 1;
        long long want = mod_seq[n-1-i];
        // Find m such that m �� want (mod mod_base) and m �� M (mod step)
        // That is, find x: m = step*x + M �� want (mod mod_base)
        // => step*x �� want-M (mod mod_base)
        long long x = solve_mod(step, (want - M + mod_base) % mod_base, mod_base);
        if (x == -1) { ok = false; break; }
        M = step * x + M;
        step *= mod_base;
    }

    if (!ok || M >= 1000000000LL) {
        cout << "NO\n";
    } else {
        cout << "YES\n" << M << endl;
    }
    return 0;
}
```
